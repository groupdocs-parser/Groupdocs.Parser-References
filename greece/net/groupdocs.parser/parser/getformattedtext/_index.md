```csharp
// Δημιουργήστε μια παρουσία της κλάσης Parser
using (Parser parser = new Parser(filePath))
{
    // Εξάγετε μορφοποιημένο κείμενο στον αναγνώστη
    using (TextReader reader = parser.GetFormattedText(new FormattedTextOptions(FormattedTextMode.Html)))
    {

    }
}
```

### Παραδείγματα

Το παρακάτω παράδειγμα δείχνει πώς να εξάγετε το κείμενο μιας σελίδας εγγράφου ως κείμενο Markdown:

```csharp
// Δημιουργήστε μια παρουσία της κλάσης Parser
using (Parser parser = new Parser(filePath))
{
    // Ελέγξτε εάν το έγγραφο υποστηρίζει εξαγωγή μορφοποιημένου κειμένου
    if (!parser.Features.FormattedText)
    {
        Console.WriteLine("Document isn't supports formatted text extraction.");
        return;
    }
    
    // Αποκτήστε τις πληροφορίες του εγγράφου
    IDocumentInfo documentInfo = parser.GetDocumentInfo();
    // Ελέγξτε αν το έγγραφο έχει σελίδες
    if (documentInfo.PageCount == 0)
    {
        Console.WriteLine("Document hasn't pages.");
        return;
    }
    
    // Διατρέξτε τις σελίδες
    for (int p = 0; p<documentInfo.PageCount; p++)
    {
        // Εκτυπώστε έναν αριθμό σελίδας 
        Console.WriteLine(string.Format("Page {0}/{1}", p + 1, documentInfo.PageCount));
        // Εξάγετε μορφοποιημένο κείμενο στον αναγνώστη
        using (TextReader reader = parser.GetFormattedText(p, new FormattedTextOptions(FormattedTextMode.Markdown)))
        {
            // Εκτυπώστε μορφοποιημένο κείμενο από το έγγραφο
            // Αγνοούμε τον έλεγχο null καθώς έχουμε ήδη ελέγξει την υποστήριξη της λειτουργίας εξαγωγής μορφοποιημένου κειμένου νωρίτερα
            Console.WriteLine(reader.ReadToEnd());
        }
    }
}
```
